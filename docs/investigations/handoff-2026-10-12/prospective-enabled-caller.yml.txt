          function appendRunnerOutput(contents) {
            let fd;
            let primaryFailure;
            try {
              const target = process.env.GITHUB_OUTPUT;
              if (typeof target !== 'string' || !target || !path.isAbsolute(target) || path.resolve(target) !== target ||
                  !/^(?:classification=(?:same-repository|cross-repository|unknown)|admitted=(?:true|false))\n$/.test(contents) ||
                  Buffer.byteLength(contents, 'utf8') > 64) throw new Error('runner output rejected');
              const before = fs.lstatSync(target);
              if (before.isSymbolicLink() || !before.isFile() || before.nlink !== 1) throw new Error('runner output rejected');
              if (fs.realpathSync(target) !== target || typeof fs.constants.O_NOFOLLOW !== 'number') throw new Error('runner output rejected');
              fd = fs.openSync(target, fs.constants.O_WRONLY | fs.constants.O_APPEND | fs.constants.O_NOFOLLOW);
              const opened = fs.fstatSync(fd);
              if (!opened.isFile() || opened.nlink !== 1 || opened.dev !== before.dev || opened.ino !== before.ino) {
                throw new Error('runner output rejected');
              }
              fs.writeFileSync(fd, contents);
            } catch (error) {
              primaryFailure = error;
              throw error;
            } finally {
              if (fd !== undefined) {
                try { fs.closeSync(fd); }
                catch (error) { if (primaryFailure === undefined) throw error; }
              }
            }
          }

          const validId = value => Number.isSafeInteger(value) && value > 0;
          let classification = 'unknown';
          try {
            const event = JSON.parse(fs.readFileSync(process.env.GITHUB_EVENT_PATH, 'utf8'));
            const rawExpected = process.env.EXPECTED_REPOSITORY_ID;
            const expected = /^[1-9][0-9]*$/.test(rawExpected) ? Number(rawExpected) : NaN;
            const rawPullRequestNumber = process.env.EXPECTED_PULL_REQUEST_NUMBER;
            const expectedPullRequestNumber = /^[1-9][0-9]*$/.test(rawPullRequestNumber)
              ? Number(rawPullRequestNumber) : NaN;
            const pullRequest = event.pull_request;
            const eventRepositoryId = event.repository?.id;
            const baseRepositoryId = pullRequest?.base?.repo?.id;
            const headRepositoryId = pullRequest?.head?.repo?.id;
            const validCommit = value => typeof value === 'string' && /^[0-9a-f]{40}$/.test(value);
            if (process.env.GITHUB_EVENT_NAME === 'pull_request_target' &&
                validId(expected) && validId(expectedPullRequestNumber) &&
                validId(eventRepositoryId) &&
                validId(baseRepositoryId) && validId(headRepositoryId) &&
                validId(pullRequest?.number) && pullRequest.number === expectedPullRequestNumber &&
                validCommit(pullRequest?.base?.sha) &&
                validCommit(pullRequest?.head?.sha) && expected === eventRepositoryId &&
                expected === baseRepositoryId) {
              classification = headRepositoryId === baseRepositoryId
                ? 'same-repository' : 'cross-repository';
            }
          } catch {
            classification = 'unknown';
          }
          try {
            appendRunnerOutput(`classification=${classification}\n`);
          } catch {
            console.error('Unable to safely update GitHub output.');
            process.exitCode = 1;
          }
          NODE

          async function regularFile(path, treeCache) {
